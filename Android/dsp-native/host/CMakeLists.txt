cmake_minimum_required(VERSION 3.22)
project(TDSpeedPortableValidation LANGUAGES CXX)
enable_testing()
add_library(tdspeed_core STATIC ../src/main/cpp/core/VarispeedEngine.cpp)
target_include_directories(tdspeed_core PUBLIC ../src/main/cpp/core)
target_compile_features(tdspeed_core PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(tdspeed_core PRIVATE /W4 /WX /fp:precise)
else()
    target_compile_options(tdspeed_core PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
add_executable(CoreTests CoreTests.cpp)
target_link_libraries(CoreTests PRIVATE tdspeed_core)
add_test(NAME PortableCoreInvariants COMMAND CoreTests)
option(TDSPEED_CORE_ASAN "Instrument the portable core and invariant tests" OFF)
if(TDSPEED_CORE_ASAN)
    foreach(target tdspeed_core CoreTests)
        if(MSVC)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
            target_link_options(${target} PRIVATE /INCREMENTAL:NO /DEBUG)
        else()
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()

if(DEFINED REFERENCE_CONFIG)

    target_link_libraries(ReferenceComparison PRIVATE tdspeed_core ${REFERENCE_LIBRARIES})
    target_compile_options(ReferenceComparison PRIVATE /fp:precise)
    add_test(NAME WindowsReferenceComparison COMMAND ReferenceComparison "${CMAKE_CURRENT_BINARY_DIR}/reference-fixtures.bin")
endif()

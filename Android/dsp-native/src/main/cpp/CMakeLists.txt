cmake_minimum_required(VERSION 3.22)
project(tdspeed_probe LANGUAGES CXX)
add_library(tdspeed_core STATIC core/VarispeedEngine.cpp)
target_compile_features(tdspeed_core PUBLIC cxx_std_17)
target_include_directories(tdspeed_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/core")
target_compile_options(tdspeed_core PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
set_target_properties(tdspeed_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(tdspeed_probe SHARED NativeBridge.cpp EngineBridge.cpp)
target_link_libraries(tdspeed_probe PRIVATE tdspeed_core)
target_compile_features(tdspeed_probe PRIVATE cxx_std_17)
target_compile_options(tdspeed_probe PRIVATE -Wall -Wextra -Werror)
target_link_options(tdspeed_probe PRIVATE "-Wl,-z,max-page-size=16384")
